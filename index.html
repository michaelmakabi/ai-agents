Method 2 — Local bridge (one-time setup)

Run a tiny Node.js bridge on your laptop. This page POSTs the plan to localhost:7777, the bridge spawns Claude Code with the right prompt, and streams stdout back here as it runs. You can switch over to Cowork or Claude Code mid-task to keep working in those environments.

Show bridge install (~30s)
1. Save this file as workforce-bridge.js in any folder:
// workforce-bridge.js — local execution bridge for the Makabi Workforce
const http = require('http');
const { spawn } = require('child_process');
const PORT = 7777;
const ALLOWED_ORIGIN = '*'; // restrict in production

http.createServer((req, res) => {
  res.setHeader('Access-Control-Allow-Origin', ALLOWED_ORIGIN);
  res.setHeader('Access-Control-Allow-Methods', 'POST, OPTIONS');
  res.setHeader('Access-Control-Allow-Headers', 'Content-Type');
  if (req.method === 'OPTIONS') return res.end();

  if (req.method === 'POST' && req.url === '/run') {
    let body = '';
    req.on('data', c => body += c);
    req.on('end', () => {
      const { prompt } = JSON.parse(body);
      res.setHeader('Content-Type', 'text/event-stream');
      res.setHeader('Cache-Control', 'no-cache');
      res.flushHeaders();
      const child = spawn('claude', [prompt], { shell: true });
      child.stdout.on('data', d => res.write(`data: ${JSON.stringify({type:'out', text:d.toString()})}\n\n`));
      child.stderr.on('data', d => res.write(`data: ${JSON.stringify({type:'err', text:d.toString()})}\n\n`));
      child.on('close', code => { res.write(`data: ${JSON.stringify({type:'done', code})}\n\n`); res.end(); });
    });
  } else {
    res.statusCode = 404; res.end();
  }
}).listen(PORT, () => console.log(`Workforce bridge listening on http://localhost:${PORT}`));
2. In that folder, run: node workforce-bridge.js
3. Below, set bridge URL to http://localhost:7777 and toggle ON.
